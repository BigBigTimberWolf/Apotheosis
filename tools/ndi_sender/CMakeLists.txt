cmake_minimum_required(VERSION 3.24)
project(ApotheosisNdiSender LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "NDI sender requires 64-bit Windows")
endif()
add_executable(ndi_sender main.cpp ../../Apotheosis/ndi/ndi_runtime.cpp)
set_target_properties(ndi_sender PROPERTIES AUTOMOC OFF AUTORCC OFF)
target_compile_features(ndi_sender PRIVATE cxx_std_20)
target_compile_definitions(ndi_sender PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(ndi_sender PRIVATE ../../Apotheosis ../../third_party/ndi/include)
target_link_libraries(ndi_sender PRIVATE d3d11 dxgi winmm)
if(MSVC)
    target_compile_options(ndi_sender PRIVATE /utf-8 /W4 /EHsc)
endif()
